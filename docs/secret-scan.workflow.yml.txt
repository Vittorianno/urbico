# MODELO para copiar para .github/workflows/secret-scan.yml
# (o acesso que usei para gravar no repositório não tem permissão para criar
#  arquivos em .github/workflows; crie este arquivo pela interface do GitHub ou
#  por um `git push` seu).

name: Secret scan

on:
  push:
  pull_request:
  workflow_dispatch:

permissions:
  contents: read

jobs:
  scan:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: actions/setup-node@v4
        with:
          node-version: 22
      - name: Arquivos versionados
        run: node scripts/check-secrets.mjs
      - name: Histórico do Git (aviso, não bloqueia)
        run: node scripts/check-secrets.mjs --history
        continue-on-error: true
